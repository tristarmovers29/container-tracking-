Step 2 — Google Apps Script (Code.gs) Apps Script mein naya project kholein, Code.gs ka purana code replace karke ye complete code paste karein. /************************************************************ * CONTAINER TRACK & TRACE * Backend: Google Apps Script * Provider: Traqo Ocean * API key is stored in Script Properties. ************************************************************/ const TRAQO_BASE_URL = "https://traqocontainer.com/api/v1"; function doGet(e) { const params = (e && e.parameter) || {}; const callback = String(params.callback || ""); let result; try { if (params.action === "health") { result = { success: true, message: "Container tracking backend is online." }; } else if (params.action === "track") { result = trackShipment_(params); } else { result = { success: false, message: "Invalid action. Use action=track or action=health." }; } } catch (err) { result = { success: false, message: err && err.message ? err.message : "Unexpected server error." }; } const output = JSON.stringify(result); // JSONP is used because the frontend is hosted on GitHub Pages. if (callback && /^[A-Za-z_$][0-9A-Za-z_$\.]*$/.test(callback)) { return ContentService .createTextOutput(callback + "(" + output + ");") .setMimeType(ContentService.MimeType.JAVASCRIPT); } return ContentService .createTextOutput(output) .setMimeType(ContentService.MimeType.JSON); } function trackShipment_(params) { const reference = String(params.referenceNumber || "") .trim() .toUpperCase(); const sealine = String(params.sealine || "") .trim() .toUpperCase(); if (!reference) { return { success: false, message: "Please enter a container or BL number." }; } if (!/^[A-Z0-9\-\/]{5,35}$/.test(reference)) { return { success: false, message: "The tracking reference format looks invalid." }; } // Carrier is required so the free plan does not rely on // automatic carrier discovery. if (!/^[A-Z0-9]{4}$/.test(sealine)) { return { success: false, message: "Please select a shipping line." }; } const apiKey = PropertiesService .getScriptProperties() .getProperty("TRAQO_API_KEY"); if (!apiKey) { return { success: false, message: "API key is not configured in Apps Script." }; } const endpoint = TRAQO_BASE_URL + "/container/" + encodeURIComponent(reference) + "?sealine=" + encodeURIComponent(sealine); const response = UrlFetchApp.fetch(endpoint, { method: "get", headers: { Authorization: "Bearer " + apiKey, Accept: "application/json" }, muteHttpExceptions: true }); const statusCode = response.getResponseCode(); const body = response.getContentText(); let data; try { data = JSON.parse(body); } catch (err) { return { success: false, message: "The tracking provider returned an unreadable response." }; } if (statusCode < 200 || statusCode >= 300 || data.success === false) { return { success: false, statusCode: statusCode, message: getProviderError_(statusCode, data) }; } const shipment = data.data || data.shipment || data; // Keep the provider's actual data; do not invent tracking events. return { success: true, referenceNumber: reference, shipment: shipment }; } function getProviderError_(statusCode, data) { const providerMessage = data.statusMessage || data.message || (data.data && data.data.message) || ""; if (statusCode === 401) { return "API key is invalid. Check Script Properties."; } if (statusCode === 402) { return "Free shipment allowance may be exhausted, or account access is paused. Check your provider dashboard. No paid upgrade has been made."; } if (statusCode === 403) { return "Enable Developer mode in the provider dashboard."; } if (statusCode === 404) { return "No shipment record was returned for this reference and shipping line. Verify both values."; } if (statusCode === 429) { return "API rate limit reached. Please wait before trying again."; } if (statusCode >= 500) { return "The tracking provider or carrier is temporarily unavailable."; } return providerMessage || "Tracking request failed (HTTP " + statusCode + ")."; } Step 3 — API key securely save karein Apps Script mein: Project Settings kholein. Script Properties tak scroll karein. Add script property par click karein. Property name: TRAQO_API_KEY Value: apni Traqo API key. Save karein. API key ko GitHub par upload na karein.
